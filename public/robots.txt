# Tele-textual — https://teletext.joaobernardo.me
#
# Files in public/ are copied to dist/ and served as themselves: Vercel checks
# the filesystem before applying the rewrites in vercel.json, so the SPA
# catch-all does not swallow this the way it swallows an unknown route.

User-agent: *
Allow: /

# The admin screens. Disallow keeps crawlers off them; it is not access control
# — /manage and /import check a signed session cookie server-side, and
# /moderator is a password form. This only keeps three screens that mean nothing
# to a visitor out of the index, and keeps crawl budget on the archive.
#
# Listed twice because the route tree is mounted once per language (see
# src/domain/routes.ts), so every path also exists under /en. The screens
# themselves are untranslated by design, but the addresses resolve.
Disallow: /moderator
Disallow: /manage
Disallow: /import
Disallow: /en/moderator
Disallow: /en/manage
Disallow: /en/import

# The API answers JSON and has nothing to index.
Disallow: /api/

# Generated by scripts/prerender.ts at build time, from the pages that actually
# exist — every address in both languages, with hreflang alternates.
Sitemap: https://teletext.joaobernardo.me/sitemap.xml
